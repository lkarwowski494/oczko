--- pasy.py
import sys
from PIL import Image
for f in sys.argv[1:]:
    im=Image.open(f).convert('RGB'); w,h=im.size; px=im.load()
    rows=[]
    for y in range(h):
        n=0; cols=[]
        for x in range(0,w,2):
            r,g,b=px[x,y]
            if r>200 and g>180 and b<170 and r-b>70: n+=1
        rows.append(n*2)
    # bands: rows with > 15% width yellow
    bands=[];start=None
    for y,n in enumerate(rows):
        on=n>w*0.12
        if on and start is None: start=y
        if not on and start is not None: bands.append((start,y-1)); start=None
    out=[]
    for a,b in bands:
        # sample colour at middle row, mean of yellow pixels
        y=(a+b)//2; s=[px[x,y] for x in range(w) if px[x,y][0]>200 and px[x,y][2]<170]
        m=tuple(sum(c[i] for c in s)//len(s) for i in range(3))
        xs=[x for x in range(w) if px[x,y][0]>200 and px[x,y][2]<170]
        out.append(f"y{a}-{b} h={b-a+1}px x{min(xs)}-{max(xs)} kolor~#{m[0]:02X}{m[1]:02X}{m[2]:02X}")
    print(f, w,'x',h, '|', '; '.join(out))
